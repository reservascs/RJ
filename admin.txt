function pularTexto(s,i){ const q=s[i]; let j=i+1; while(j<s.length){ if(s[j]==='\\'){j+=2;continue;} if(s[j]===q) return j+1; j++; } return j; }
function lerPacotes(js){
  const k=js.indexOf('const PACOTES'); if(k<0) throw new Error('const PACOTES não encontrado');
  const ini=js.indexOf('[',k); let i=ini+1, prof=0, cur=-1; const itens=[];
  while(i<js.length){
    const c=js[i];
    if(c==='"'||c==="'"||c==='`'){ i=pularTexto(js,i); continue; }
    if(c==='/'&&js[i+1]==='/'){ const n=js.indexOf('\n',i); i=n<0?js.length:n; continue; }
    if(c==='/'&&js[i+1]==='*'){ i=js.indexOf('*/',i)+2; continue; }
    if(c==='{'||c==='['){ if(prof===0&&c==='{') cur=i; prof++; }
    else if(c==='}'||c===']'){
      if(prof===0&&c===']') return {itens, fim:i};
      prof--;
      if(prof===0&&c==='}'&&cur>=0){ itens.push({a:cur,b:i+1,obj:(new Function('return ('+js.slice(cur,i+1)+');'))()}); cur=-1; }
    }
    i++;
  }
  throw new Error('fim do array PACOTES não encontrado');
}

}
